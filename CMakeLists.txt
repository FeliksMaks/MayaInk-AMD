cmake_minimum_required(VERSION 3.21)
project(inkling2_mini_rdna3 LANGUAGES C CXX HIP)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_HIP_STANDARD 17)
if(NOT CMAKE_HIP_ARCHITECTURES)
  set(CMAKE_HIP_ARCHITECTURES "gfx1100")
endif()
find_package(hip CONFIG REQUIRED)
find_package(hipblas CONFIG REQUIRED)
set(COMPAT ${CMAKE_CURRENT_SOURCE_DIR}/include/strata/hip_compat)
set(IK2 ${CMAKE_CURRENT_SOURCE_DIR}/src/ik2)
set(IK2_FILES ${IK2}/ik2_kernels.cu ${IK2}/ik2_main.cu ${IK2}/ik2_draft.cu)
set_source_files_properties(${IK2_FILES} PROPERTIES LANGUAGE HIP)
add_executable(maya-ik2 ${IK2_FILES} ${IK2}/ik2_cpu.cpp)
set_source_files_properties(${IK2}/ik2_cpu.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c;-ffp-contract=off")
target_include_directories(maya-ik2 BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/amd/compat ${COMPAT}
                           ${CMAKE_CURRENT_SOURCE_DIR}/include ${IK2} ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml)
target_compile_definitions(maya-ik2 PRIVATE STRATA_USE_HIP=1 IK2_HIP=1 STRATA_HAS_WMMA=1 NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(maya-ik2 PRIVATE "SHELL:-include \"${COMPAT}/cuda_runtime.h\""
                       "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/amd/compat/ik2_hip_rt.h\"")
target_compile_options(maya-ik2 PRIVATE "$<$<COMPILE_LANGUAGE:HIP>:-Wno-unused-result;-Wno-deprecated-declarations>")
# (no build-machine paths in the binary: __FILE__ relative to the source tree)
target_compile_options(maya-ik2 PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}/=")
target_link_libraries(maya-ik2 PRIVATE hip::host roc::hipblas)
